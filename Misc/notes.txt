# BOOTSTRAPPED UCB
# No pre-training

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import SGDClassifier
from sklearn.metrics import average_precision_score
from contextualbandits.online import BootstrappedUCB
import time
import os
import warnings

# Ignore minor warnings from sklearn if a batch has only 1 class
warnings.filterwarnings('ignore')

# ==========================================
# 1. SETUP AND CONFIGURATION
# ==========================================
DATA_PATH = r"D:\Credit Card Fraud Detection\Data\creditcard_processed.csv"
RESULTS_DIR = r"D:\Credit Card Fraud Detection\results"
RESULTS_FILE = os.path.join(RESULTS_DIR, "cmab_bootstrappeducb_results1.csv")

BATCH_SIZE = 2000

# Reward Matrix (Mapped to business logic)
REWARD_TP = 20     # Blocked fraud (+20)
REWARD_TN = 1      # Approved good customer (+1)
REWARD_FP = -10    # Declined good customer (-10 friction)
REWARD_FN = -100   # Approved fraud (-100 lost money)

# Training augment weight
FRAUD_WEIGHT = 100 

os.makedirs(RESULTS_DIR, exist_ok=True)

# ==========================================
# 2. LOAD DATA
# ==========================================
print("Loading preprocessed data...")
df = pd.read_csv(DATA_PATH)
X = df.drop(columns=['Class'])
y = df['Class']

# ==========================================
# 3. INITIALIZE THE CONTEXTUAL BANDIT
# ==========================================
base_algorithm = SGDClassifier(loss='log_loss', random_state=42)

# Bootstrapped UCB Initialization
bandit = BootstrappedUCB(
    base_algorithm=base_algorithm,
    nchoices=2,
    nsamples=10,            # Creates 10 model clones per action arm
    percentile=80,          # Uses the 80th percentile prediction as the Upper Bound
    batch_train=True,       # Required to enable partial_fit for streaming
    batch_sample_method='gamma', # Standard stable weights for online bootstrapping
    random_state=42
)

# ==========================================
# 4. STREAMING SIMULATION
# ==========================================
print("\nStarting Streaming Simulation (Bootstrapped UCB)...")
num_batches = len(df) // BATCH_SIZE + (1 if len(df) % BATCH_SIZE != 0 else 0)

# Trackers for our required metrics
cumulative_reward = 0
cumulative_regret = 0

# Trackers for PR-AUC 
historical_y_true = []
historical_y_score = []

simulation_results = []
start_time = time.time()

for batch_idx in range(num_batches):
    start_row = batch_idx * BATCH_SIZE
    end_row = min((batch_idx + 1) * BATCH_SIZE, len(df))
    
    X_batch = X.iloc[start_row:end_row]
    y_batch = y.iloc[start_row:end_row].values
    
    # --- A. PREDICTIONS ---
    # bandit.predict looks at the 80th percentile of the 10 clones to choose an action
    actions = bandit.predict(X_batch.values, exploit=False)
    
    try:
        scores = bandit.decision_function(X_batch.values)
        batch_scores = scores[:, 1]
    except Exception:
        # Fallback for the very first batch before models are trained
        batch_scores = np.ones(len(y_batch)) * 0.5

    historical_y_true.extend(y_batch)
    historical_y_score.extend(batch_scores)
    
    # --- B. REWARD & REGRET CALCULATION ---
    batch_reward = 0
    
    for i in range(len(y_batch)):
        if y_batch[i] == 1:
            if actions[i] == 1: batch_reward += REWARD_TP
            else: batch_reward += REWARD_FN
        else:
            if actions[i] == 0: batch_reward += REWARD_TN
            else: batch_reward += REWARD_FP
            
    optimal_batch_reward = np.sum((y_batch == 1) * REWARD_TP + (y_batch == 0) * REWARD_TN)
    batch_regret = optimal_batch_reward - batch_reward
    
    cumulative_reward += batch_reward
    cumulative_regret += batch_regret
    
    if sum(historical_y_true) > 0:
        cum_pr_auc = average_precision_score(historical_y_true, historical_y_score)
    else:
        cum_pr_auc = 0.0

    # --- C. SAVE METRICS ---
    simulation_results.append({
        "Batch": batch_idx + 1,
        "Cumulative_Reward": cumulative_reward,               
        "Cumulative_Regret": cumulative_regret,               
        "Cumulative_PR_AUC": cum_pr_auc,                      
        "Fraud_in_Batch": sum(y_batch == 1)
    })
    
    # --- D. BANDIT UPDATE ---
    binary_rewards = (actions == y_batch).astype(int)
    
    is_fraud_mask = (y_batch == 1)
    repeats = np.where(is_fraud_mask, FRAUD_WEIGHT, 1)
    
    X_fit = np.repeat(X_batch.values, repeats, axis=0)
    a_fit = np.repeat(actions, repeats, axis=0)
    r_fit = np.repeat(binary_rewards, repeats, axis=0)
    
    bandit.partial_fit(X_fit, a_fit, r_fit)
    
    if (batch_idx + 1) % 10 == 0:
        print(f"Batch {batch_idx+1:03d}/{num_batches} | Cum. Reward: {cumulative_reward:<8} | Regret: {cumulative_regret:<8} | PR-AUC: {cum_pr_auc:.4f}")

print(f"\nSimulation complete in {time.time() - start_time:.2f} seconds.")

# ==========================================
# 5. EXPORT RESULTS
# ==========================================
results_df = pd.DataFrame(simulation_results)
results_df.to_csv(RESULTS_FILE, index=False)
print(f"Evaluation metrics securely saved to: {RESULTS_FILE}")

# ==========================================
# 6. VISUALIZATION DASHBOARD
# ==========================================
plt.figure(figsize=(18, 5))

# Plot 1: Cumulative Reward
plt.subplot(1, 3, 1)
plt.plot(results_df['Batch'], results_df['Cumulative_Reward'], color='green', linewidth=2)
plt.title('Cumulative Reward')
plt.xlabel('Batch Number')
plt.ylabel('Cumulative Reward ($)')
plt.grid(True)

# Plot 2: Cumulative Regret 
plt.subplot(1, 3, 2)
plt.plot(results_df['Batch'], results_df['Cumulative_Regret'], color='red', linewidth=2)
plt.title('Cumulative Regret')
plt.xlabel('Batch Number')
plt.ylabel('Cumulative Regret ($)')
plt.grid(True)

# Plot 3: Cumulative PR-AUC
plt.subplot(1, 3, 3)
plt.plot(results_df['Batch'], results_df['Cumulative_PR_AUC'], color='blue', linewidth=2)
plt.title('Cumulative PR-AUC')
plt.xlabel('Batch Number')
plt.ylabel('Area Under Precision-Recall Curve')
plt.grid(True)

plt.tight_layout()
plt.show()

import pandas as pd

REWARD_FP = -10    # Declined good customer (-10 friction)
REWARD_FN = -100   # Approved fraud (-100 lost money)

# Training augment weight
FRAUD_WEIGHT = 100 

os.makedirs(RESULTS_DIR, exist_ok=True)

# 2. LOAD & SPLIT DATA (20% Warm-up / 80% Live)
# ==========================================
print("Loading preprocessed data...")
df = pd.read_csv(DATA_PATH)

# Split exactly like the Supervised Learning models
train_size = int(len(df) * 0.20)
train_df = df.iloc[:train_size]
test_df = df.iloc[train_size:].reset_index(drop=True)

X_train = train_df.drop(columns=['Class']).values
y_train = train_df['Class'].values

X_test = test_df.drop(columns=['Class'])
y_test = test_df['Class']

# Calculate the starting batch number so it aligns perfectly with the Supervised plots

# ==========================================

# ==========================================
base_algorithm = SGDClassifier(loss='log_loss', random_state=42)

bandit = BootstrappedUCB(
    base_algorithm=base_algorithm,
    nchoices=2,
    nsamples=10,            
    percentile=80,          
    batch_train=True,       
    batch_sample_method='gamma', 
    random_state=42
)

# ==========================================
# 4. PHASE 1: WARM START (OFFLINE TRAINING)
# ==========================================
print(f"\n[PHASE 1] Warm Starting Bandit on first 20% of data ({len(train_df)} rows)...")
num_warmup_batches = len(train_df) // BATCH_SIZE + (1 if len(train_df) % BATCH_SIZE != 0 else 0)

start_time = time.time()

for batch_idx in range(num_warmup_batches):
    start_row = batch_idx * BATCH_SIZE
    end_row = min((batch_idx + 1) * BATCH_SIZE, len(train_df))
    
    X_batch = X_train[start_row:end_row]
    y_batch = y_train[start_row:end_row]
    
    # Bandit explores and makes predictions on historical data
    actions = bandit.predict(X_batch, exploit=False)
    
    # Calculate binary rewards (0/1) for internal math
    binary_rewards = (actions == y_batch).astype(int)
    
    # Augment and train (But DO NOT track business rewards/regret yet!)
    is_fraud_mask = (y_batch == 1)
    repeats = np.where(is_fraud_mask, FRAUD_WEIGHT, 1)
    
    X_fit = np.repeat(X_batch, repeats, axis=0)
    a_fit = np.repeat(actions, repeats, axis=0)
    r_fit = np.repeat(binary_rewards, repeats, axis=0)
    
    bandit.partial_fit(X_fit, a_fit, r_fit)

print(f"Warm Start complete in {time.time() - start_time:.2f} seconds. Bandit is now trained and ready!")

# ==========================================
# 5. PHASE 2: LIVE DEPLOYMENT (STREAMING EVALUATION)
# ==========================================
print(f"\n[PHASE 2] Live Deployment on remaining 80% of data (Starting at Batch {starting_batch_number})...")
num_test_batches = len(test_df) // BATCH_SIZE + (1 if len(test_df) % BATCH_SIZE != 0 else 0)

        cum_pr_auc = average_precision_score(historical_y_true, historical_y_score)

        cum_pr_auc = 0.0

    current_batch_number = starting_batch_number + batch_idx

    # --- C. SAVE METRICS ---
    simulation_results.append({

REWARD_TP = 20     # Blocked fraud (+20)
REWARD_TN = 1      # Approved good customer (+1)
REWARD_FP = -10    # Declined good customer (-10 friction)
REWARD_FN = -100   # Approved fraud (-100 lost money)

# Training augment weight
FRAUD_WEIGHT = 100

    base_algorithm=base_algorithm,
    nchoices=2,             
    explore_prob=0.2,       
    decay=0.9999,           
    batch_train=True,       
    random_state=42
)

REWARD_TP = 20     # Blocked fraud (+20)
REWARD_TN = 1      # Approved good customer (+1)
REWARD_FP = -10    # Declined good customer (-10 friction)
REWARD_FN = -100   # Approved fraud (-100 lost money)

X_train = train_df.drop(columns=['Class']).values
y_train = train_df['Class'].values

X_test = test_df.drop(columns=['Class'])
y_test = test_df['Class']

starting_batch_number = (train_size // BATCH_SIZE) + 1

bandit = LinUCB(

    alpha=1.0,         # The exploration parameter

    random_state=42
)

# ==========================================

    start_row = batch_idx * BATCH_SIZE
    end_row = min((batch_idx + 1) * BATCH_SIZE, len(train_df))
    
    X_batch = X_train[start_row:end_row]
    y_batch = y_train[start_row:end_row]
    
    # Bandit explores and makes predictions on historical data
    actions = bandit.predict(X_batch, exploit=False)
    
    # Calculate binary rewards (0/1) for internal math
    binary_rewards = (actions == y_batch).astype(int)
    
    # Augment and train (But DO NOT track business rewards/regret yet!)
    is_fraud_mask = (y_batch == 1)
    repeats = np.where(is_fraud_mask, FRAUD_WEIGHT, 1)

    a_fit = np.repeat(actions, repeats, axis=0)
    r_fit = np.repeat(binary_rewards, repeats, axis=0)
    
    bandit.partial_fit(X_fit, a_fit, r_fit)

import pandas as pd

# ==========================================
# 1. SETUP AND CONFIGURATION
# ==========================================
DATA_PATH = r"D:\Credit Card Fraud Detection\Data\creditcard_processed.csv"
RESULTS_DIR = r"D:\Credit Card Fraud Detection\Results"
RESULTS_FILE = os.path.join(RESULTS_DIR, "cmab_lints_results5.csv")

# Reward Matrix (Mapped to business logic)
REWARD_TP = 20     # Blocked fraud (+20)
REWARD_TN = 1      # Approved good customer (+1)
REWARD_FP = -10    # Declined good customer (-10 friction)
REWARD_FN = -100   # Approved fraud (-100 lost money)

# Training augment weight
FRAUD_WEIGHT = 10 # used to be 100, but reduced to 10 for LinTS to avoid overfitting    

os.makedirs(RESULTS_DIR, exist_ok=True)

# ==========================================
# 2. LOAD DATA
# ==========================================
print("Loading preprocessed data...")
df = pd.read_csv(DATA_PATH)
X = df.drop(columns=['Class'])
y = df['Class']

# ==========================================
# 3. INITIALIZE THE CONTEXTUAL BANDIT (LinTS)
# ==========================================
# Notice: No base_algorithm. It uses internal Bayesian Linear Regression.

bandit = LinTS(
    nchoices=2,
    lambda_=10.0,        # NEW: Regularization (The Brakes)
    v_sq=0.01,            # Variance parameter (Controls the width of the Bayesian distribution)
    sample_unique=True,  # Mathematically correct: Samples a unique coefficient vector for EVERY row
    method='sm',         # Sherman-Morrison formula for efficient sequential batch updates
    random_state=42
)

# ==========================================
# 4. STREAMING SIMULATION
# ==========================================
print("\nStarting Streaming Simulation (LinTS)...")
print("NOTE: LinTS calculates Eigenvalues for every row. This will be slower than Bootstrapping!")

num_batches = len(df) // BATCH_SIZE + (1 if len(df) % BATCH_SIZE != 0 else 0)

# Trackers for our required metrics
cumulative_reward = 0
cumulative_regret = 0

# Trackers for PR-AUC 
historical_y_true = []
historical_y_score = []

simulation_results = []
start_time = time.time()

for batch_idx in range(num_batches):
    start_row = batch_idx * BATCH_SIZE
    end_row = min((batch_idx + 1) * BATCH_SIZE, len(df))
    
    X_batch = X.iloc[start_row:end_row]
    y_batch = y.iloc[start_row:end_row].values
    
    # --- A. PREDICTIONS ---
    # bandit.predict samples from the Bayesian posterior distribution
    actions = bandit.predict(X_batch.values, exploit=False)
    
    try:
        # decision_function outputs the expected score before sampling
        scores = bandit.decision_function(X_batch.values)
        batch_scores = scores[:, 1]
    except Exception:
        # Fallback for the very first batch before the matrix is initialized
        batch_scores = np.ones(len(y_batch)) * 0.5

    historical_y_true.extend(y_batch)

        if y_batch[i] == 1:
            if actions[i] == 1: batch_reward += REWARD_TP
            else: batch_reward += REWARD_FN
        else:
            if actions[i] == 0: batch_reward += REWARD_TN
            else: batch_reward += REWARD_FP    
        batch_reward = max(0, batch_reward)
    
    optimal_batch_reward = np.sum((y_batch == 1) * REWARD_TP + (y_batch == 0) * REWARD_TN)
    batch_regret = optimal_batch_reward - batch_reward
    cumulative_reward += batch_reward
    cumulative_regret += batch_regret
    
    if sum(historical_y_true) > 0:
        cum_pr_auc = average_precision_score(historical_y_true, historical_y_score)
    else:
        cum_pr_auc = 0.0

    # --- C. SAVE METRICS ---
    simulation_results.append({
        "Batch": batch_idx + 1,
        "Cumulative_Reward": cumulative_reward,               
        "Cumulative_Regret": cumulative_regret,               
        "Cumulative_PR_AUC": cum_pr_auc,                      
        "Fraud_in_Batch": sum(y_batch == 1)
    })
    
    # --- D. BANDIT UPDATE ---
    binary_rewards = (actions == y_batch).astype(int)
    
    is_fraud_mask = (y_batch == 1)
    repeats = np.where(is_fraud_mask, FRAUD_WEIGHT, 1)
    
    X_fit = np.repeat(X_batch.values, repeats, axis=0)
    a_fit = np.repeat(actions, repeats, axis=0)
    r_fit = np.repeat(binary_rewards, repeats, axis=0)

    bandit.partial_fit(X_fit, a_fit, r_fit)
    
    if (batch_idx + 1) % 10 == 0:
        print(f"Batch {batch_idx+1:03d}/{num_batches} | Cum. Reward: {cumulative_reward:<8} | Regret: {cumulative_regret:<8} | PR-AUC: {cum_pr_auc:.4f}")

print(f"\nSimulation complete in {time.time() - start_time:.2f} seconds.")
